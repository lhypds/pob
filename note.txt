
Commands
--------

Rebuild and install and launch
pob kill && ./build.sh && ./install.sh && pob launch

Open a Microsandbox instance and connect with TigerVNC
pob launch --msb --vncviewer

Start a pob session in a Microsandbox instance
msb exec msb-b400 -- pob start

TigerVNC connect
vncviewer 127.0.0.1:5901

Microsandbox
msb stop xxx
msb rm xxx

Delete all Microsandbox instances
msb list -q | grep '^msb-' | xargs msb rm -f

Rebuild install and lanch a Sandbox
./build.sh && ./install.sh && pob launch --msb --vncviewer

Launch a sandbox with fullscreen
pob launch --msb --vncviewer --fullscreen

Create a new instance
pob new
pob new "Work laptop"

kill all pob instances and delete them all
pob kill; pob kill --all; for id in $(ls ~/.pob 2>/dev/null | grep '^pb-'); do pob del "$id" --yes; done
